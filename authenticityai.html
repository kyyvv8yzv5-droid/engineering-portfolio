In plain language

When the evidence pipeline cannot apply its policy, the result explicitly says that the decision is inconclusive.

Inputs, outputs and control flow

Accepts an error reason. Builds a FusionResult with neutral scores, zero confidence and no contributions, then adds the compatibility fields consumed by callers.

Design, integration and security

The engine keeps the public response shape even when policy or inputs are invalid. The neutral score is a sentinel and should not be interpreted as a calibrated probability. Fusion weighting and model configuration are omitted.

def decode_image(data):
    return cv2.imdecode(
        np.frombuffer(data, np.uint8),
        cv2.IMREAD_COLOR,
    )


def load_and_sample_video(data, extension):
    temporary_path = None

    try:
        with NamedTemporaryFile(
            suffix=extension,
            delete=False,
            dir=MANAGED_PATHS.temp if MANAGED_PATHS.configured else None,
        ) as temporary_file:
            temporary_file.write(data)
            temporary_path = Path(temporary_file.name)

        capture = cv2.VideoCapture(str(temporary_path))

        if not capture.isOpened():
            return None

        try:
            fps = float(capture.get(cv2.CAP_PROP_FPS))
            frame_count = int(
                capture.get(cv2.CAP_PROP_FRAME_COUNT)
            )
            width = int(
                capture.get(cv2.CAP_PROP_FRAME_WIDTH)
            )
            height = int(
                capture.get(cv2.CAP_PROP_FRAME_HEIGHT)
            )
            duration = frame_count / fps if fps > 0 else 0

            sample_count = min(12, frame_count)
            sampled_frames = []

            if frame_count > 0 and sample_count > 0:
                frame_indices = np.linspace(
                    0,
                    frame_count - 1,
                    num=sample_count,
                    dtype=int,
                )

                for frame_index in frame_indices:
                    capture.set(
                        cv2.CAP_PROP_POS_FRAMES,
                        int(frame_index),
                    )
                    success, frame = capture.read()

                    if success and frame is not None:
                        sampled_frames.append(frame)

            return {
                "fps": fps,
                "frame_count": frame_count,
                "width": width,
                "height": height,
                "duration": duration,
                "sampled_frames": sampled_frames,
            }
        finally:
            capture.release()
    finally:
        if temporary_path is not None:
            temporary_path.unlink(missing_ok=True)

def _clean_value(value):
    if value is None:
        return None
    if isinstance(value, bytes):
        value = value.decode("utf-8", errors="replace")
    if isinstance(value, str):
        value = value.strip().strip("\x00")
        return value or None
    return value


def _software_warnings(software):
    if not software:
        return []

    normalized = str(software).casefold()
    warnings = []

    if any(marker in normalized for marker in EDITING_SOFTWARE_MARKERS):
        warnings.append("editing_software_detected")

    if any(
        marker in normalized
        for marker in GENERATOR_OR_AI_SOFTWARE_MARKERS
    ):
        warnings.append("generator_or_ai_software_detected")

    return warnings

def inspect_image_metadata(data):
    try:
        with Image.open(io.BytesIO(data)) as image:
            exif = image.getexif()
            info = image.info

            camera_make = _clean_value(exif.get(271))
            camera_model = _clean_value(exif.get(272))
            software = _clean_value(
                exif.get(305)
                or info.get("Software")
                or info.get("software")
            )

            datetime_fields = [
                _clean_value(exif.get(306)),
                _clean_value(exif.get(36867)),
                _clean_value(exif.get(36868)),
            ]
            capture_datetime = (
                datetime_fields[1]
                or datetime_fields[2]
                or datetime_fields[0]
                or _clean_value(
                    info.get("creation_time")
                    or info.get("Creation Time")
                    or info.get("date:create")
                )
            )

            gps_present = False
            try:
                gps_present = bool(exif.get_ifd(34853))
            except (KeyError, TypeError, ValueError):
                gps_present = bool(exif.get(34853))

            orientation = _clean_value(exif.get(274))
            color_space_value = _clean_value(exif.get(40961))

            if color_space_value == 1:
                color_space = "sRGB"
            elif color_space_value == 65535:
                color_space = "Uncalibrated"
            elif color_space_value is not None:
                color_space = str(color_space_value)
            elif info.get("icc_profile"):
                color_space = "ICC profile embedded"
            else:
                color_space = None

            descriptive_info_present = any(
                info.get(key)
                for key in (
                    "Software",
                    "software",
                    "creation_time",
                    "Creation Time",
                    "date:create",
                    "icc_profile",
                    "exif",
                )
            )
            metadata_present = bool(exif) or descriptive_info_present

    except (OSError, UnidentifiedImageError, ValueError):
        return {
            "metadata_present": False,
            "camera_make": None,
            "camera_model": None,
            "software": None,
            "capture_datetime": None,
            "gps_present": False,
            "orientation": None,
            "color_space": None,
            "metadata_warnings": [
                "metadata_inspection_failed",
                NO_METADATA_CONTEXT,
            ],
        }

    warnings = _software_warnings(software)
    populated_datetimes = {
        value for value in datetime_fields if value is not None
    }

    if len(populated_datetimes) > 1:
        warnings.append("inconsistent_datetime_fields")

    # Missing metadata is common after ordinary platform processing. It must
    # never be treated as evidence that an image was AI-generated.
    if not metadata_present:
        warnings.extend(["no_metadata", NO_METADATA_CONTEXT])

    return {
        "metadata_present": bool(metadata_present),
        "camera_make": camera_make,
        "camera_model": camera_model,
        "software": software,
        "capture_datetime": capture_datetime,
        "gps_present": gps_present,
        "orientation": orientation,
        "color_space": color_space,
        "metadata_warnings": warnings,
    }


def inspect_video_metadata(data, extension, ffprobe_path=None):
    ffprobe = ffprobe_path or shutil.which("ffprobe")

    if not ffprobe:
        return _empty_video_metadata(
            ["ffprobe_unavailable", NO_METADATA_CONTEXT]
        )

    temporary_path = None

    try:
        with NamedTemporaryFile(
            suffix=extension,
            delete=False,
            dir=MANAGED_PATHS.temp if MANAGED_PATHS.configured else None,
        ) as temporary_file:
            temporary_file.write(data)
            temporary_path = Path(temporary_file.name)

        completed = subprocess.run(
            [
                ffprobe,
                "-v",
                "error",
                "-print_format",
                "json",
                "-show_format",
                "-show_streams",
                str(temporary_path),
            ],
            capture_output=True,
            check=True,
            text=True,
            timeout=15,
        )
        probe = json.loads(completed.stdout)
    except (
        FileNotFoundError,
        json.JSONDecodeError,
        OSError,
        subprocess.CalledProcessError,
        subprocess.TimeoutExpired,
    ):
        return _empty_video_metadata(
            ["metadata_inspection_failed", NO_METADATA_CONTEXT]
        )
    finally:
        if temporary_path is not None:
            temporary_path.unlink(missing_ok=True)

    format_data = probe.get("format") or {}
    streams = probe.get("streams") or []
    video_stream = next(
        (
            stream
            for stream in streams
            if stream.get("codec_type") == "video"
        ),
        {},
    )
    format_tags = format_data.get("tags") or {}
    stream_tags = video_stream.get("tags") or {}

    format_name = _clean_value(format_data.get("format_name"))
    codec_name = _clean_value(video_stream.get("codec_name"))
    encoder = _clean_value(
        format_tags.get("encoder") or stream_tags.get("encoder")
    )
    creation_time = _clean_value(
        format_tags.get("creation_time")
        or stream_tags.get("creation_time")
    )
    bitrate_value = _clean_value(format_data.get("bit_rate"))

    try:
        bitrate = int(bitrate_value) if bitrate_value is not None else None
    except (TypeError, ValueError):
        bitrate = bitrate_value

    metadata_present = any(
        value is not None
        for value in (
            format_name,
            codec_name,
            encoder,
            creation_time,
            bitrate,
        )
    )
    warnings = _software_warnings(encoder)

    if not metadata_present:
        warnings.extend(["no_metadata", NO_METADATA_CONTEXT])

    return {
        "format_name": format_name,
        "codec_name": codec_name,
        "encoder": encoder,
        "creation_time": creation_time,
        "bitrate": bitrate,
        "metadata_present": metadata_present,
        "metadata_warnings": warnings,
    }

import cv2
import numpy as np


def analyze_image_quality(image):
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

    brightness = float(np.mean(gray))
    contrast = float(np.std(gray))
    sharpness = float(
        cv2.Laplacian(gray, cv2.CV_64F).var()
    )

    warnings = []

    if brightness < 40:
        warnings.append("Very dark")

    if brightness > 220:
        warnings.append("Overexposed")

    if sharpness < 100:
        warnings.append("Blurry")

    if contrast < 25:
        warnings.append("Low contrast")

    return {
        "brightness": round(brightness, 2),
        "contrast": round(contrast, 2),
        "sharpness": round(sharpness, 2),
    }, warnings


def analyze_video_quality(frames):
    sharpness_scores = []
    brightness_scores = []

    for frame in frames:
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        sharpness_scores.append(
            float(
                cv2.Laplacian(
                    gray,
                    cv2.CV_64F,
                ).var()
            )
        )
        brightness_scores.append(float(np.mean(gray)))

    return {
        "average_brightness": (
            round(float(np.mean(brightness_scores)), 2)
            if brightness_scores
            else None
        ),
        "average_sharpness": (
            round(float(np.mean(sharpness_scores)), 2)
            if sharpness_scores
            else None
        ),
    }

import json
import logging
import os
import time
from contextlib import contextmanager


# Uvicorn configures this logger in both normal and reload workers. Using its
# handler keeps development diagnostics visible without changing production
# logging configuration or adding response fields.
LOGGER = logging.getLogger("uvicorn.error")


def development_timing_enabled():
    return os.getenv(
        "AUTHENTICITY_DEV_TIMING",
        "false",
    ).strip().lower() == "true"


def log_timing(stage, duration_seconds, **details):
    if not development_timing_enabled():
        return
    LOGGER.info(
        json.dumps(
            {
                "event": "analysis_timing",
                "stage": stage,
                "duration_ms": round(duration_seconds * 1000.0, 3),
                **details,
            },
            sort_keys=True,
        )
    )


@contextmanager
def timed_stage(stage, **details):
    started = time.perf_counter()
    try:
        yield
    finally:
        log_timing(stage, time.perf_counter() - started, **details)

from collections import OrderedDict
import time

from detectors.learned.aide import AIDEDetector
from detectors.learned.clip_sid import CLIPSIDDetector
from detectors.learned.ssp import SSPDetector
from detectors.learned.trufor import TruForDetector
from detectors.learned.universal_fake import UniversalFakeDetector


class LearnedDetectorRegistry:
    """Ordered, fail-isolated collection of independent learned detectors."""

    def __init__(self, detectors=None):
        if detectors is None:
            detectors = [
                UniversalFakeDetector(),
                AIDEDetector(),
                CLIPSIDDetector(),
                SSPDetector(),
                TruForDetector(),
            ]
        self._detectors = OrderedDict((detector.key, detector) for detector in detectors)

    @classmethod
    def default(cls, universal_fake_detector=None):
        return cls([
            universal_fake_detector or UniversalFakeDetector(),
            AIDEDetector(),
            CLIPSIDDetector(),
            SSPDetector(),
            TruForDetector(),
        ])

    def get(self, key):
        return self._detectors[key]

    def keys(self):
        return self._detectors.keys()

    def analyze_image(self, image, cached=None):
        cached = cached or {}
        results = OrderedDict()
        for key, detector in self._detectors.items():
            if key in cached:
                results[key] = cached[key]
                continue
            started = time.perf_counter()
            try:
                value = detector.analyze_image(image).to_dict()
            except Exception as error:
                result = detector._result(
                    warnings=["experimental_model", "detector_internal_failure"],
                    error=f"{type(error).__name__}: {error}",
                )
                value = result.to_dict()
            results[key] = self._evidence_contract(
                key, value, (time.perf_counter() - started) * 1000.0
            )
        return dict(results)

    def analyze_video(self, cached=None):
        cached = cached or {}
        results = OrderedDict()
        for key, detector in self._detectors.items():
            if key in cached:
                results[key] = cached[key]
                continue
            started = time.perf_counter()
            try:
                value = detector.video_result().to_dict()
            except Exception as error:
                result = detector._result(
                    applicable=False,
                    warnings=["experimental_model", "detector_internal_failure"],
                    error=f"{type(error).__name__}: {error}",
                )
                value = result.to_dict()
            results[key] = self._evidence_contract(
                key, value, (time.perf_counter() - started) * 1000.0
            )
        return dict(results)

    @staticmethod
    def _evidence_contract(key, value, elapsed_ms):
        value = dict(value)
        probability = value.get("ai_probability")
        available = bool(value.get("available")) and probability is not None
        confidence = min(1.0, 0.55 + abs(float(probability) - 0.5) * 0.8) if available else 0.0
        direction = (
            "supports_ai" if available and probability >= 0.58 else
            "supports_authentic" if available and probability <= 0.42 else
            "neutral"
        )
        value["confidence"] = round(confidence, 6)
        value["explanation"] = (
            "Local learned inference completed; confidence is an uncalibrated distance-from-0.5 heuristic, not empirical accuracy."
            if available else
            "The local learned detector was unavailable, disabled, or not applicable."
        )
        value["evidence"] = [] if not available else [{
            "evidence_id": f"{key}.model_probability",
            "category": "learned_detector",
            "direction": direction,
            "strength": round(abs(float(probability) - 0.5) * 2.0, 6),
            "confidence": round(confidence, 6),
            "description": "Uncalibrated sigmoid output from a locally supplied, fail-closed learned detector.",
            "metrics": {"ai_probability": round(float(probability), 6)},
        }]
        value["processing_time_ms"] = round(max(0.0, elapsed_ms), 3)
        if not value.get("inference_time_ms"):
            value["inference_time_ms"] = value["processing_time_ms"]
        return value

"""Stable contracts shared by every offline Evidence Engine detector."""

from __future__ import annotations

import time
from dataclasses import asdict, dataclass, field
from typing import Any

import cv2
import numpy as np


def clamp(value, minimum=0.0, maximum=1.0):
    return float(max(minimum, min(maximum, float(value))))


@dataclass(frozen=True)
class EvidenceObject:
    evidence_id: str
    category: str
    direction: str
    strength: float
    confidence: float
    description: str
    metrics: dict[str, Any] = field(default_factory=dict)

    def to_dict(self):
        value = asdict(self)
        value["strength"] = round(clamp(value["strength"]), 6)
        value["confidence"] = round(clamp(value["confidence"]), 6)
        return value


@dataclass
class DetectorResult:
    detector_id: str
    applicable: bool
    ai_probability: float | None
    confidence: float
    explanation: str
    evidence: list[EvidenceObject] = field(default_factory=list)
    processing_time_ms: float = 0.0
    weight: float = 1.0
    reliability: float = 0.5
    false_positive_rate: float = 0.1
    warnings: list[str] = field(default_factory=list)
    error: str | None = None

    def to_dict(self):
        return {
            "detector_id": self.detector_id,
            "applicable": bool(self.applicable),
            "ai_probability": (
                None if self.ai_probability is None
                else round(clamp(self.ai_probability), 6)
            ),
            "confidence": round(clamp(self.confidence), 6),
            "explanation": self.explanation,
            "evidence": [item.to_dict() for item in self.evidence],
            "processing_time_ms": round(max(0.0, self.processing_time_ms), 3),

def contribution_sentence(row):
    probability = row.normalized_ai_probability
    if probability is None:
        return None
    direction = "AI-generated origin" if probability > 0.5 else "authentic origin"

def build_explanations(contributions, media_reasons, conflict_ids):
    ranked = sorted((row for row in contributions if row.effective_weight > 0), key=lambda row: row.effective_weight * abs((row.normalized_ai_probability or 0.5) - 0.5), reverse=True)
    supporting = [sentence for row in ranked if (sentence := contribution_sentence(row))][:6]

    limitations = list(dict.fromkeys(media_reasons))
    unavailable = [row.detector_name for row in contributions if row.status in {"unavailable", "disabled", "unsupported", "failed"}]

        limitations.append(f"Unavailable, disabled, unsupported, or failed sources did not vote: {', '.join(unavailable[:8])}.")

"""Binary, calibration, grouped, and bootstrap evaluation metrics."""

from __future__ import annotations

import math
import random
from collections import defaultdict


def _validate(scores, labels):
    scores = [max(0.0, min(1.0, float(value))) for value in scores]
    labels = [1 if value in (1, True, "ai") else 0 for value in labels]
    if not scores or len(scores) != len(labels):
        raise ValueError("scores and labels must be non-empty and equally sized")
    return scores, labels


def confusion(scores, labels, threshold=0.5):
    scores, labels = _validate(scores, labels)
    predictions = [int(score >= threshold) for score in scores]
    return {
        "tp": sum(p == y == 1 for p, y in zip(predictions, labels)),
        "tn": sum(p == y == 0 for p, y in zip(predictions, labels)),
        "fp": sum(p == 1 and y == 0 for p, y in zip(predictions, labels)),
        "fn": sum(p == 0 and y == 1 for p, y in zip(predictions, labels)),
    }


def _safe(numerator, denominator):
    return numerator / denominator if denominator else 0.0


def roc_auc(scores, labels):
    scores, labels = _validate(scores, labels)
    positives = sum(labels)
    negatives = len(labels) - positives
    if not positives or not negatives:
        return None
    ordered = sorted(zip(scores, labels), key=lambda row: row[0])
    rank_sum = 0.0
    index = 0
    while index < len(ordered):
        end = index + 1
        while end < len(ordered) and ordered[end][0] == ordered[index][0]:
            end += 1
        average_rank = (index + 1 + end) / 2.0
        rank_sum += average_rank * sum(label for _score, label in ordered[index:end])
        index = end
    return (rank_sum - positives * (positives + 1) / 2.0) / (positives * negatives)


def pr_auc(scores, labels):
    scores, labels = _validate(scores, labels)
    positives = sum(labels)
    if not positives:
        return None
    ordered = sorted(zip(scores, labels), reverse=True)
    tp = fp = 0
    previous_recall = 0.0
    area = 0.0
    for _score, label in ordered:
        tp += label
        fp += 1 - label
        recall = tp / positives
        precision = tp / (tp + fp)
        area += (recall - previous_recall) * precision
        previous_recall = recall
    return area


def expected_calibration_error(scores, labels, bins=10):
    scores, labels = _validate(scores, labels)
    total = len(scores)
    result = 0.0
    reliability = []
    for index in range(bins):
        low, high = index / bins, (index + 1) / bins
        selected = [position for position, score in enumerate(scores) if low <= score < high or (index == bins - 1 and score == 1.0)]
        if not selected:
            reliability.append({"low": low, "high": high, "count": 0, "mean_confidence": None, "observed_rate": None})
            continue
        confidence = sum(scores[position] for position in selected) / len(selected)
        observed = sum(labels[position] for position in selected) / len(selected)
        result += len(selected) / total * abs(confidence - observed)
        reliability.append({"low": low, "high": high, "count": len(selected), "mean_confidence": confidence, "observed_rate": observed})
    return result, reliability


def binary_metrics(scores, labels, threshold=0.5):
    scores, labels = _validate(scores, labels)
    matrix = confusion(scores, labels, threshold)
    tp, tn, fp, fn = matrix["tp"], matrix["tn"], matrix["fp"], matrix["fn"]
    precision = _safe(tp, tp + fp)
    recall = _safe(tp, tp + fn)
    specificity = _safe(tn, tn + fp)
    ece, reliability = expected_calibration_error(scores, labels)
    return {
        "sample_count": len(scores),
        "accuracy": _safe(tp + tn, len(scores)),
        "balanced_accuracy": (recall + specificity) / 2.0,
        "precision": precision,
        "recall": recall,
        "specificity": specificity,
        "f1": _safe(2 * precision * recall, precision + recall),
        "false_positive_rate": _safe(fp, fp + tn),
        "false_negative_rate": _safe(fn, fn + tp),
        "roc_auc": roc_auc(scores, labels),
        "pr_auc": pr_auc(scores, labels),
        "brier_score": sum((score - label) ** 2 for score, label in zip(scores, labels)) / len(scores),
        "expected_calibration_error": ece,
        "confusion_matrix": matrix,
        "reliability_diagram": reliability,
        "threshold": threshold,
    }


def bootstrap_confidence_intervals(scores, labels, *, groups=None, iterations=500, seed=2026, confidence=0.95):
    scores, labels = _validate(scores, labels)
    rng = random.Random(seed)
    if groups is None:
        units = [[index] for index in range(len(scores))]
    else:
        if len(groups) != len(scores):
            raise ValueError("groups must match score count")
        by_group = defaultdict(list)
        for index, group in enumerate(groups):
            by_group[str(group)].append(index)
        units = list(by_group.values())
    values = defaultdict(list)
    for _ in range(int(iterations)):
        selected = [rng.choice(units) for _unit in units]
        indices = [index for unit in selected for index in unit]
        sample_labels = [labels[index] for index in indices]
        if len(set(sample_labels)) < 2:
            continue
        result = binary_metrics([scores[index] for index in indices], sample_labels)
        for name in ("accuracy", "balanced_accuracy", "precision", "recall", "specificity", "f1", "false_positive_rate", "false_negative_rate", "roc_auc", "pr_auc", "brier_score", "expected_calibration_error"):
            if result[name] is not None:
                values[name].append(result[name])
    alpha = (1.0 - confidence) / 2.0
    output = {}
    for name, samples in values.items():
        samples.sort()
        if not samples:
            continue
        low = samples[max(0, math.floor(alpha * (len(samples) - 1)))]
        high = samples[min(len(samples) - 1, math.ceil((1 - alpha) * (len(samples) - 1)))]
        output[name] = {"low": low, "high": high, "confidence": confidence, "successful_iterations": len(samples)}
    return output


def grouped_metrics(rows, field, threshold=0.5):
    grouped = defaultdict(list)
    for row in rows:
        grouped[str(row.get(field) or "unknown")].append(row)
    return {
        name: binary_metrics(
            [row["score"] for row in members],
            [row["label"] for row in members],
            threshold,
        )
        for name, members in grouped.items()
        if len({row["label"] for row in members}) == 2
    }


def confidence_band_metrics(rows, threshold=0.5, bands=((0.0, 0.5), (0.5, 0.75), (0.75, 0.9), (0.9, 1.01))):
    output = {}
    for low, high in bands:
        members = [row for row in rows if low <= float(row.get("confidence", 0.0)) < high]
        summary = {
            "count": len(members),
            "accuracy": _safe(sum(row.get("correct", False) for row in members), len(members)),
        }
        if members and len({row["label"] for row in members}) == 2:
            summary.update(binary_metrics([row["score"] for row in members], [row["label"] for row in members], threshold))
        output[f"{low:.2f}-{min(high, 1.0):.2f}"] = summary
    return output

from __future__ import annotations

def _bucket(value, seed):
    digest = hashlib.sha256(f"{seed}|{value}".encode("utf-8")).digest()
    return int.from_bytes(digest[:8], "big") / 2**64


def assign_standard_splits(records, validation_fraction=0.2, test_fraction=0.2, seed=2026):
    if validation_fraction <= 0 or test_fraction <= 0 or validation_fraction + test_fraction >= 1:
        raise ValueError("validation and test fractions must be positive and sum to less than one")
    family_by_sample, _families, _relations = build_families(records)
    output = []
    for record in records:
        value = _bucket(family_by_sample[record["sample_id"]], seed)
        split = "test" if value < test_fraction else "validation" if value < test_fraction + validation_fraction else "train"
        output.append({**record, "split": split})
    return output


def _family_name(record, field, fallback):
    value = record.get(field) or record.get(fallback)
    if not value:
        return None
    return str(value).split(":", 1)[0].strip().casefold()


def leave_one_family_out(records, kind, seed=2026):
    if kind == "generator":
        candidates = sorted({_family_name(record, "generator_family", "generator") for record in records if record.get("label") == "ai"} - {None})
        field, fallback, label = "generator_family", "generator", "ai"
    elif kind == "camera":
        candidates = sorted({_family_name(record, "camera_family", "camera_device") for record in records if record.get("label") == "real"} - {None})
        field, fallback, label = "camera_family", "camera_device", "real"
    else:
        raise ValueError("kind must be generator or camera")
    suites = []
    family_by_sample, _families, _relations = build_families(records)
    for held_out in candidates:
        case = []
        for record in records:
            family = _family_name(record, field, fallback)
            if record.get("label") == label and family == held_out:
                split = "test"
            else:
                value = _bucket(family_by_sample[record["sample_id"]], seed)
                if record.get("label") != label and value < 0.2:
                    split = "test"
                elif value < 0.4:
                    split = "validation"
                else:
                    split = "train"
            case.append({**record, "split": split})
        suites.append({"held_out": held_out, "records": case})
    return suites

import json
import tempfile
import unittest
from pathlib import Path

from fusion import EvidenceContribution, FusionEngine, FusionPolicyError, load_policy
import web_ui


def learned(probability=None, *, available=True, enabled=True, confidence=0.9):
    return {
        "universal_fake_detect": {
            "enabled": enabled,
            "available": available,
            "ai_probability": probability,
            "raw_score": None if probability is None else probability * 4 - 2,
            "confidence": confidence,
            "calibration_status": "uncalibrated",
            "warnings": [],
        }
    }


def forensic(detector_id, probability, *, confidence=0.9, reliability=0.8, warnings=None):
    return {
        detector_id: {
            "applicable": True,
            "ai_probability": probability,
            "confidence": confidence,
            "reliability": reliability,
            "warnings": list(warnings or []),
            "explanation": "Synthetic test signal.",
        }
    }


class FusionTests(unittest.TestCase):
    def setUp(self):
        self.engine = FusionEngine()
        self.context = {
            "metadata": {},
            "compression": {"applicable": True},
            "noise": {"applicable": True, "noise_warnings": []},
            "quality": {"sharpness": 500, "brightness": 128, "contrast": 50},
            "quality_warnings": [],
            "face_count": 0,
            "resolution": {"width": 1024, "height": 1024},
            "media_type": "image",
        }

    def run_fusion(self, learned_detectors=None, forensic_detectors=None, **changes):
        context = dict(self.context)
        context.update(changes)
        return self.engine.analyze(
            learned_detectors=learned_detectors or {},
            forensic_detectors=forensic_detectors or {},
            **context,
        )

    def test_policy_loads_and_is_versioned(self):
        policy = load_policy()
        self.assertEqual(policy["schema_version"], 1)
        self.assertIn("strong_min_coverage", policy["gates"])

    def test_contribution_contract_rejects_unknown_status(self):
        with self.assertRaises(ValueError):
            EvidenceContribution("x", "X", "1", "test", True, True, "negative")

    def test_disabled_detector_is_not_negative_evidence(self):
        result = self.run_fusion(learned(False, available=False, enabled=False))
        row = next(item for item in result["detector_contributions"] if item["detector_id"] == "universal_fake_detect")
        self.assertEqual(row["status"], "disabled")
        self.assertEqual(row["effective_weight"], 0.0)
        self.assertEqual(result["verdict"], "inconclusive")
        self.assertEqual(result["ai_score"], 0.5)

    def test_failed_detector_has_zero_weight(self):
        source = forensic("future_failed", 0.99)
        source["future_failed"]["error"] = "inference failed"
        result = self.run_fusion(forensic_detectors=source)
        row = next(item for item in result["contributions"] if item["detector_id"] == "future_failed")
        self.assertEqual(row["status"], "failed")
        self.assertEqual(row["effective_weight"], 0.0)

    def test_unsupported_detector_has_zero_weight(self):
        source = {"future_video_only": {"applicable": False, "ai_probability": 0.99}}
        result = self.run_fusion(forensic_detectors=source)
        row = next(item for item in result["contributions"] if item["detector_id"] == "future_video_only")
        self.assertEqual(row["status"], "unsupported")
        self.assertEqual(row["effective_weight"], 0.0)

    def test_all_learned_unavailable_cannot_produce_strong_verdict(self):
        result = self.run_fusion(learned(None, available=False), forensic("diffusion_edge_artifacts", 0.99))
        self.assertNotIn(result["verdict"], {"likely_ai_generated", "likely_authentic"})
        self.assertIn("No learned image detector contributed", result["limitations"][0])

    def test_one_uncalibrated_learned_detector_never_produces_strong_verdict(self):
        result = self.run_fusion(learned(0.99))
        self.assertNotIn(result["verdict"], {"likely_ai_generated", "likely_authentic"})
        row = next(item for item in result["detector_contributions"] if item["detector_id"] == "universal_fake_detect")
        self.assertEqual(row["status"], "uncalibrated")
        self.assertEqual(row["calibration_status"], "uncalibrated")

    def test_strong_multifamily_ai_agreement(self):
        result = self.run_fusion(learned(0.96), forensic("diffusion_edge_artifacts", 0.91))
        self.assertEqual(result["verdict"], "likely_ai_generated")
        self.assertGreater(result["weighted_ai_evidence"], 0)

    def test_strong_multifamily_authentic_agreement(self):
        result = self.run_fusion(learned(0.04), forensic("prnu_approximation", 0.08))
        self.assertEqual(result["verdict"], "likely_authentic")
        self.assertGreater(result["weighted_authentic_evidence"], 0)

    def test_conflicting_models_are_named_and_reduce_confidence(self):
        agreeing = self.run_fusion(learned(0.92), forensic("diffusion_edge_artifacts", 0.88))
        conflict = self.run_fusion(learned(0.92), forensic("diffusion_edge_artifacts", 0.08))
        self.assertGreater(conflict["conflict_score"], 0)
        self.assertIn("universal_fake_detect", conflict["conflicting_detectors"])
        self.assertIn("diffusion_edge_artifacts", conflict["conflicting_detectors"])
        self.assertLess(conflict["confidence"], agreeing["confidence"])

    def test_high_conflict_forces_inconclusive(self):
        opposing = {}
        for detector_id in ("diffusion_edge_artifacts", "prnu_approximation", "gan_checkerboard_artifacts", "lighting_consistency"):
            opposing.update(forensic(detector_id, 0.02, confidence=1.0, reliability=1.0))
        result = self.run_fusion(learned(0.98, confidence=1.0), opposing)
        self.assertGreater(result["conflict_score"], 0.70)
        self.assertEqual(result["verdict"], "inconclusive")

    def test_poor_media_quality_blocks_strong_verdict(self):
        result = self.run_fusion(
            learned(0.98), forensic("diffusion_edge_artifacts", 0.95),
            resolution={"width": 64, "height": 64}, quality_warnings=["Blurry", "Very dark"],
            compression={"applicable": True, "estimated_jpeg_quality": 20},
        )
        self.assertNotIn(result["verdict"], {"likely_ai_generated", "likely_authentic"})
        self.assertTrue(any("resolution" in item.lower() for item in result["limitations"]))

    def test_low_resolution_reduces_weight(self):
        high = self.run_fusion(learned(0.9), forensic("diffusion_edge_artifacts", 0.9))
        low = self.run_fusion(learned(0.9), forensic("diffusion_edge_artifacts", 0.9), resolution={"width": 120, "height": 120})
        def weight(result):
            return next(item["effective_weight"] for item in result["contributions"] if item["detector_id"] == "universal_fake_detect")
        self.assertLess(weight(low), weight(high))

    def test_low_coverage_forces_inconclusive(self):
        result = self.run_fusion(forensic_detectors=forensic("future_detector", 0.99, confidence=0.01, reliability=0.01))
        self.assertEqual(result["verdict"], "inconclusive")

    def test_missing_metadata_is_neutral(self):
        result = self.run_fusion()
        row = next(item for item in result["detector_contributions"] if item["detector_id"] == "metadata_provenance")
        self.assertEqual(row["status"], "inconclusive")
        self.assertIsNone(row["normalized_ai_probability"])
        self.assertEqual(row["effective_weight"], 0.0)

    def test_generator_metadata_is_directional_but_not_sufficient_for_strong_verdict(self):
        result = self.run_fusion(metadata={"metadata_warnings": ["generator_or_ai_software_detected"]})
        row = next(item for item in result["detector_contributions"] if item["detector_id"] == "metadata_provenance")
        self.assertGreater(row["normalized_ai_probability"], 0.5)
        self.assertNotIn(result["verdict"], {"likely_ai_generated", "likely_authentic"})

    def test_warnings_reduce_effective_weight(self):
        clear = self.run_fusion(forensic_detectors=forensic("diffusion_edge_artifacts", 0.9))
        warned = self.run_fusion(forensic_detectors=forensic("diffusion_edge_artifacts", 0.9, warnings=["a", "b"]))
        def weight(result):
            return next(item["effective_weight"] for item in result["detector_contributions"] if item["detector_id"] == "diffusion_edge_artifacts")
        self.assertLess(weight(warned), weight(clear))

    def test_false_negative_learned_signal_is_exposed_as_conflict(self):
        others = forensic("diffusion_edge_artifacts", 0.94)
        others.update(forensic("gan_checkerboard_artifacts", 0.90))
        result = self.run_fusion(learned(0.194), others)
        self.assertGreater(result["conflict_score"], 0)
        self.assertIn("universal_fake_detect", result["conflicting_detectors"])

    def test_universal_fake_strong_positive_requires_independent_support(self):
        result = self.run_fusion(learned(0.99), forensic("diffusion_edge_artifacts", 0.95))
        self.assertEqual(result["verdict"], "likely_ai_generated")

    def test_future_detector_uses_generic_adapter(self):
        result = self.run_fusion(forensic_detectors=forensic("future_detector_v2", 0.8))
        row = next(item for item in result["contributions"] if item["detector_id"] == "future_detector_v2")
        self.assertEqual(row["evidence_family"], "forensic")
        self.assertEqual(row["normalized_ai_probability"], 0.8)
        self.assertIn("timing_ms", row)

    def test_deterministic_repeated_results_except_timing(self):
        first = self.run_fusion(learned(0.8), forensic("prnu_approximation", 0.7))
        second = self.run_fusion(learned(0.8), forensic("prnu_approximation", 0.7))
        first.pop("processing_time_ms")
        second.pop("processing_time_ms")
        self.assertEqual(first, second)

    def test_invalid_policy_fails_closed(self):
        with tempfile.TemporaryDirectory() as directory:
            path = Path(directory) / "bad.json"
            path.write_text(json.dumps({"schema_version": 99}), encoding="utf-8")
            result = FusionEngine(policy_path=path).analyze()
        self.assertEqual(result["verdict"], "inconclusive")
        self.assertEqual(result["confidence"], 0.0)
        self.assertIsNotNone(result["error"])

    def test_video_is_inconclusive_when_image_detectors_are_unsupported(self):
        result = self.run_fusion(learned(None, available=False), media_type="video")
        self.assertEqual(result["verdict"], "inconclusive")

    def test_ui_contains_dedicated_fusion_card_and_disclaimer(self):
        rendered = web_ui.render_analyze_page()
        self.assertIn("Evidence fusion summary", rendered)
        self.assertIn("Probabilistic evidence summary", rendered)
        self.assertIn("renderFusion(fields,body.fusion)", rendered)

    def test_explanation_and_requested_alias_contract(self):
        result = self.run_fusion(learned(0.9), forensic("diffusion_edge_artifacts", 0.85))
        for field in ("primary_reasons", "supporting_reasons", "counter_evidence", "limitations", "warnings", "contributions"):
            self.assertIn(field, result)
            self.assertIsInstance(result[field], list)

Overview and problem

AuthenticityAI is a local research platform for evaluating media authenticity signals. It addresses the need to inspect heterogeneous evidence while keeping provenance, detector availability, and uncertainty visible to reviewers.

Architecture and integrations

A Python and FastAPI service hosts an operator UI and an isolated showcase. OpenCV and MediaPipe support analysis, SQLite and a managed filesystem hold local state, and a fusion layer combines detector evidence. Separate acquisition, training, and evaluation workflows use governed dataset sources. The research path includes optional learned detector adapters and an offline CNN baseline.

Implemented capabilities and safeguards

The platform supports media analysis, provenance review, duplicate handling, dataset acquisition, controlled splits, and evidence reports. Local authentication and role checks separate operators from showcase access. Detector failures and insufficient evidence remain explicit. The model research described here does not establish commercial decision accuracy.

Engineering challenge

Detector outputs have different reliability and availability. The system needs to preserve provenance and uncertainty through normalization, evidence fusion, and reporting.

Validation and roadmap

The source-only snapshot passed 472 Python and 28 JavaScript tests with disposable SQLite data and the OpenCV fallback. External research data and learned weights were outside this verification.
