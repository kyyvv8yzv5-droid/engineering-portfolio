async def upload_evidence(
    conversation_id: str, files: list[UploadFile] = File(...), restaurant_id: str | None = None,
    x_rmh_user: str = Header(default="local-user"), workspace_id: str = Depends(get_workspace_id),
) -> dict:
    if not files or len(files) > settings.evidence_max_files_per_upload:
        raise HTTPException(413, detail={"code": "too_many_files", "message": f"Choose up to {settings.evidence_max_files_per_upload} files at a time."})
    results: list[dict] = []; total = 0
    for upload in files:
        data = await upload.read(settings.evidence_max_file_mb * 1024 * 1024 + 1)
        total += len(data)
        if total > settings.evidence_max_upload_mb * 1024 * 1024:
            results.append({"filename": upload.filename or "file", "mime_type": upload.content_type, "size": len(data), "status": "failed", "message": f"The total upload must be {settings.evidence_max_upload_mb} MB or smaller."})
            continue
        try:
            results.append(evidence_intake.accept(workspace_id=workspace_id, user_id=x_rmh_user[:100], conversation_id=conversation_id,
                restaurant_id=restaurant_id, filename=upload.filename or "file", declared_mime=upload.content_type, data=data))
        except evidence_intake.EvidenceIntakeError as exc:
            results.append({"filename": upload.filename or "file", "mime_type": upload.content_type, "size": len(data),
                            "status": "unsupported" if exc.status == 415 else "failed", "message": str(exc)})
        finally:
            await upload.close()
    accepted = sum(item["status"] == "ready_for_review" for item in results)
    return {"files": results, "accepted": accepted, "rejected": len(results) - accepted}

def _decode_bundle(row: sqlite3.Row, items: list[sqlite3.Row], restaurant: Any | None) -> dict[str, Any]:
    locations: dict[int, dict[str, Any]] = {}
    item_payloads: list[dict[str, Any]] = []
    for item in items:
        value = json.loads(item["structured_value_json"])
        locations.setdefault(item["entity_index"], {})[item["field_name"]] = value
        item_payloads.append({
            "id": item["id"], "entity_index": item["entity_index"], "field_name": item["field_name"],
            "value": value, "raw_supplied_text": item["raw_supplied_text"], "confidence": item["confidence"],
            "confidence_basis": item["confidence_basis"], "verification_status": item["verification_status"],
            "source_references": json.loads(item["source_references_json"] or "[]"),
            "provenance_chain": json.loads(item["provenance_chain_json"] or "[]"),
        })
    existing = {_address_key(location.model_dump()): location for location in (restaurant.locations if restaurant else []) if _address_key(location.model_dump())}
    duplicates = []
    conflicts = []
    seen: dict[str, int] = {}
    for index, location in locations.items():
        key = _address_key(location)
        if key and key in existing:
            duplicates.append({"location_index": index, "existing_location_id": existing[key].id, "message": "This address is already saved."})
        if key and key in seen:

        elif key:
            seen[key] = index
        if not location.get("address_line_1") and not location.get("formatted_address"):
            conflicts.append({"location_index": index, "field_name": "address", "message": "Add an address before saving this location."})
    payload = dict(row)

def _zip_mime(data: bytes, extension: str) -> str | None:
    try:
        with zipfile.ZipFile(BytesIO(data)) as archive:
            names = set(archive.namelist())
            if extension == ".docx" and "word/document.xml" in names: return ZIP_MIMES[extension]
            if extension == ".xlsx" and "xl/workbook.xml" in names: return ZIP_MIMES[extension]
            if extension == ".pptx" and "ppt/presentation.xml" in names: return ZIP_MIMES[extension]
            if extension in {".odt", ".ods", ".odp"} and "mimetype" in names:
                value = archive.read("mimetype").decode("ascii", "ignore").strip()
                return value if value == ZIP_MIMES[extension] else None
    except (zipfile.BadZipFile, KeyError, OSError): pass

def detect_mime(data: bytes, extension: str) -> str:
    head = data[:512]
    if head.startswith(b"\xff\xd8\xff"): return "image/jpeg"
    if head.startswith(b"\x89PNG\r\n\x1a\n"): return "image/png"
    if head.startswith(b"RIFF") and head[8:12] == b"WEBP": return "image/webp"
    if len(head) >= 12 and head[4:8] == b"ftyp" and head[8:12] in {b"heic", b"heix", b"hevc", b"hevx", b"mif1", b"msf1"}: return "image/heic" if extension == ".heic" else "image/heif"
    if head.startswith(b"%PDF-"): return "application/pdf"
    if head.startswith(b"{\\rtf"): return "application/rtf"
    if head.startswith(bytes.fromhex("D0CF11E0A1B11AE1")): return "application/x-ole-storage"
    if head.startswith(b"PK\x03\x04"): return _zip_mime(data, extension) or "application/zip"
    sample = data[:8192]
    if b"\x00" in sample: return "application/octet-stream"
    try: text = sample.decode("utf-8")
    except UnicodeDecodeError: return "application/octet-stream"
    stripped = text.lstrip()
    if extension == ".json":
        try: json.loads(text); return "application/json"
        except json.JSONDecodeError: return "text/plain"
    if extension in {".xml"} and stripped.startswith("<?xml"): return "application/xml"
    if extension in {".html", ".htm"} and re.search(r"(?i)<(?:!doctype\s+html|html|head|body)\b", stripped): return "text/html"
    if extension == ".csv" and ("," in text or "\t" in text): return "text/csv"
    if extension == ".md": return "text/markdown"
    return "text/plain"


def _safe_text(data: bytes, extension: str, *, preserve_lines: bool = False) -> str | None:
    if extension in {".txt", ".md", ".csv", ".json", ".xml", ".html", ".htm", ".rtf"}:
        value = data.decode("utf-8", "replace")
    elif extension in ZIP_MIMES:
        try:
            with zipfile.ZipFile(BytesIO(data)) as archive:
                candidates = [name for name in archive.namelist() if name.endswith(".xml") and name.startswith(("word/", "xl/", "ppt/slides/", "content.xml"))]
                value = "\n".join(archive.read(name).decode("utf-8", "replace") for name in candidates[:100])
        except (zipfile.BadZipFile, OSError): return None
    else: return None
    value = re.sub(r"(?is)<script\b.*?</script>|<style\b.*?</style>", " ", value)
    value = html.unescape(re.sub(r"<[^>]+>", " ", value))
    value = re.sub(r"[\x00-\x08\x0b\x0c\x0e-\x1f]", "", value)
    if preserve_lines:
        value = re.sub(r"[ \t]+", " ", value)
        value = re.sub(r"\s*\n\s*", "\n", value)
        value = re.sub(r"\n{3,}", "\n\n", value)

function Inline({ text }: { text: string }) {
  const parts = text.split(/(`[^`]+`|\*\*[^*]+\*\*)/g);
  return <>{parts.map((part, index) => part.startsWith("**") ? <strong key={index}>{part.slice(2, -2)}</strong> : part.startsWith("`") ? <code key={index} className="rounded bg-black/5 px-1 py-0.5 text-[.9em]">{part.slice(1, -1)}</code> : part)}</>;

function Markdown({ content }: { content: string }) {
  const lines = sanitizeDisplay(content).replace(/\r/g, "").split("\n");
  const nodes: React.ReactNode[] = [];
  let code: string[] = [];
  let inCode = false;
  lines.forEach((line, index) => {
    if (line.startsWith("```")) {
      if (inCode) { nodes.push(<pre key={`code-${index}`} className="my-2 overflow-x-auto rounded-xl bg-[#27232e] p-3 text-xs text-white"><code>{code.join("\n")}</code></pre>); code = []; }
      inCode = !inCode; return;
    }
    if (inCode) { code.push(line); return; }
    if (/^\d+\.\s/.test(line)) nodes.push(<div key={index} className="mt-1.5 flex gap-2"><span className="font-semibold text-brand">{line.match(/^\d+/)?.[0]}.</span><p><Inline text={line.replace(/^\d+\.\s*/, "")}/></p></div>);
    else if (/^[-*]\s/.test(line)) nodes.push(<div key={index} className="mt-1 flex gap-2"><span aria-hidden>•</span><p><Inline text={line.slice(2)}/></p></div>);
    else if (line.startsWith("### ")) nodes.push(<h4 key={index} className="mt-3 font-semibold"><Inline text={line.slice(4)}/></h4>);
    else if (line.startsWith("## ")) nodes.push(<h3 key={index} className="mt-3 text-sm font-semibold"><Inline text={line.slice(3)}/></h3>);
    else if (line.trim()) nodes.push(<p key={index} className="mt-1.5 first:mt-0"><Inline text={line}/></p>);
  });
  return <div className="break-words text-[13px] leading-[1.55]">{nodes}</div>;
}

Overview and problem

Culinary IQ supports restaurant operations teams working with media, menus, and location information. Its assistant brings relevant application context and product guidance into the operator workflow and makes proposed updates reviewable.

Architecture and integrations

A Next.js, React, and TypeScript dashboard uses a Python and FastAPI backend. The assistant integrates OpenAI tool calling with contextual retrieval over a versioned product manual. SQLite persists conversations, tool audits, and pending evidence. Picker and an optional mapping provider support evidence collection through bounded integrations.

Implemented capabilities and safeguards

The workspace supports a persistent assistant, contextual tools, restaurant onboarding, menu and media workflows, and field-level evidence review. Pending evidence records its source and can be rejected. Approved updates pass through explicit confirmation. The current development identity is fixed, so external deployment requires authenticated workspace membership and tenant-isolation verification.

Engineering challenge

Assistant proposals combine user input, app records, and provider evidence. The workflow has to preserve field provenance and give the operator control over canonical updates.

The sanitized snapshot passed 217 backend and eight frontend tests plus a Next.js production build. Authentication and tenant isolation remain deployment gates.
