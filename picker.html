    func recognize(_ data: Data) async throws -> [RecognizedTextItem] {
        try Task.checkCancellation()
        guard let source = CGImageSourceCreateWithData(data as CFData, nil),
              let image = CGImageSourceCreateImageAtIndex(source, 0, nil) else {
            throw LocalOCRError.unsupportedImage
        }
        let properties = CGImageSourceCopyPropertiesAtIndex(source, 0, nil) as? [CFString: Any]
        let rawOrientation = (properties?[kCGImagePropertyOrientation] as? NSNumber)?.uint32Value ?? 1
        let orientation = CGImagePropertyOrientation(rawValue: rawOrientation) ?? .up
        let request = VNRecognizeTextRequest()
        request.recognitionLevel = .accurate
        request.usesLanguageCorrection = true
        return try await withTaskCancellationHandler {
            try Task.checkCancellation()
            try VNImageRequestHandler(cgImage: image, orientation: orientation).perform([request])
            try Task.checkCancellation()
            return (request.results ?? []).compactMap { observation in
                guard let candidate = observation.topCandidates(1).first else { return nil }

                    confidence: candidate.confidence,
                    boundingBox: TextBoundingBox(x: Double(box.minX), y: Double(box.minY),
                                                width: Double(box.width), height: Double(box.height))
                )
            }
        } onCancel: {
            request.cancel()
        }
    }

import hashlib

from PIL import Image


def perceptual_hash(image: Image.Image) -> str:
    sample = image.convert("L").resize((9, 8), Image.Resampling.LANCZOS)
    pixels = list(sample.getdata())
    value = 0
    for row in range(8):
        for column in range(8):
            value = (value << 1) | int(pixels[row * 9 + column] > pixels[row * 9 + column + 1])
    return f"{value:016x}"


def calculate_hashes(image_bytes: bytes, image: Image.Image | None = None) -> tuple[str, str]:
    if image is None:
        from io import BytesIO
        with Image.open(BytesIO(image_bytes)) as source:
            image = source.convert("RGB").copy()
    return hashlib.sha256(image_bytes).hexdigest(), perceptual_hash(image)


def compare_exact_hashes(left: str, right: str) -> bool:
    return left == right


def compare_perceptual_hashes(left: str, right: str) -> int:
    return (int(left, 16) ^ int(right, 16)).bit_count()

def _safe_trim(image: Image.Image) -> tuple[Image.Image, tuple[int, int, int, int]]:
    width, height = image.size
    corners = [image.getpixel(point) for point in ((0, 0), (width - 1, 0), (0, height - 1), (width - 1, height - 1))]
    background_color = tuple(sorted(pixel[channel] for pixel in corners)[len(corners) // 2] for channel in range(3))
    background = Image.new("RGB", image.size, background_color)
    difference = ImageChops.difference(image, background).convert("L").point(lambda value: 255 if value > 14 else 0)
    box = difference.getbbox()
    if not box:
        return image, (0, 0, width, height)
    left, top, right, bottom = box
    retained = ((right - left) * (bottom - top)) / (width * height)
    if retained < 0.60 or (left < width * .01 and top < height * .01 and right > width * .99 and bottom > height * .99):
        return image, (0, 0, width, height)
    padding = max(2, round(min(width, height) * .004))
    box = (max(0, left - padding), max(0, top - padding), min(width, right + padding), min(height, bottom + padding))
    return image.crop(box), box


def _encode(image: Image.Image, maximum_edge: int, quality: int) -> tuple[bytes, tuple[int, int]]:
    normalized = image.copy()
    normalized.thumbnail((maximum_edge, maximum_edge), Image.Resampling.LANCZOS)
    if normalized.size != image.size:
        normalized = normalized.filter(ImageFilter.UnsharpMask(radius=.7, percent=115, threshold=2))
    output = BytesIO()
    normalized.save(output, "JPEG", quality=quality, optimize=True)
    return output.getvalue(), normalized.size


def normalize_screenshot(image_bytes: bytes, config: PickerConfig | None = None, *, maximum_edge: int | None = None) -> PreparedScreenshot:
    config = config or PickerConfig()
    if len(image_bytes) > config.maximum_upload_bytes:
        raise ValueError("Screenshot exceeds the configured maximum upload size")
    started = time.perf_counter()
    try:
        with Image.open(BytesIO(image_bytes)) as source:
            oriented = ImageOps.exif_transpose(source)
            if "A" in oriented.getbands():
                alpha_box = oriented.getchannel("A").getbbox()
                oriented = oriented.crop(alpha_box) if alpha_box else oriented
            original = oriented.convert("RGB").copy()
    except (UnidentifiedImageError, OSError, ValueError) as exc:
        raise ValueError("The screenshot could not be decoded as an image") from exc
    trimmed, trim_box = _safe_trim(original)

    return _encode(prepared.trimmed, maximum_edge, quality)


def validate_crop(box: PickerBoundingBoxV1, minimum_area: float = .03) -> bool:
    return box.x + box.width <= 1.001 and box.y + box.height <= 1.001 and box.width * box.height >= minimum_area


def refine_crop_to_visible_edges(image: Image.Image, box: PickerBoundingBoxV1) -> PickerBoundingBoxV1:
    """Snap a model crop to nearby strong horizontal media/UI boundaries."""
    if box.width < .75:
        return box
    sample = image.convert("L")
    sample.thumbnail((360, 900), Image.Resampling.BOX)
    scores = [0.0]
    for y in range(1, sample.height):
        difference = ImageChops.difference(sample.crop((0, y - 1, sample.width, y)), sample.crop((0, y, sample.width, y + 1)))
        scores.append(ImageStat.Stat(difference).mean[0])
    def strongest(center: float, radius: float) -> tuple[float, float]:
        start = max(1, round((center - radius) * sample.height)); end = min(sample.height - 1, round((center + radius) * sample.height))
        index = max(range(start, end + 1), key=lambda item: scores[item])
        return index / sample.height, scores[index]
    top, top_score = strongest(box.y, .07)
    bottom, bottom_score = strongest(box.y + box.height, .16)
    if top_score < 10 or bottom_score < 10 or bottom - top < .2:
        return box
    return PickerBoundingBoxV1(x=box.x, y=top, width=box.width, height=bottom - top)


def remap_box(box: PickerBoundingBoxV1 | None, prepared: PreparedScreenshot) -> PickerBoundingBoxV1 | None:
    if box is None:
        return None
    left, top, right, bottom = prepared.trim_box
    original_width, original_height = prepared.original.size
    trimmed_width, trimmed_height = right - left, bottom - top
    return PickerBoundingBoxV1(
        x=(left + box.x * trimmed_width) / original_width,
        y=(top + box.y * trimmed_height) / original_height,
        width=box.width * trimmed_width / original_width,
        height=box.height * trimmed_height / original_height,
    )


def crop_content_region(image_bytes: bytes, box: PickerBoundingBoxV1, *, quality: int = 92) -> tuple[bytes, tuple[int, int]]:
    if not validate_crop(box):
        raise ValueError("Crop coordinates are invalid")
    with Image.open(BytesIO(image_bytes)) as source:
        image = ImageOps.exif_transpose(source).convert("RGB")
        left = max(0, min(image.width - 1, round(box.x * image.width)))
        top = max(0, min(image.height - 1, round(box.y * image.height)))
        right = max(left + 1, min(image.width, round((box.x + box.width) * image.width)))
        bottom = max(top + 1, min(image.height, round((box.y + box.height) * image.height)))
        crop = image.crop((left, top, right, bottom))
    output = BytesIO(); crop.save(output, "JPEG", quality=quality, optimize=True)
    return output.getvalue(), crop.size

def _fallback_reasons(value: CompactExtraction, config: PickerConfig) -> list[str]:
    reasons: list[str] = []
    for name, field in (("platform", value.pf), ("caption", value.caption)):
        if not field.v: reasons.append(f"missing_{name}")
        elif field.c < config.required_field_confidence_threshold: reasons.append(f"low_{name}_confidence")
    crop = _box(value.image.b)
    if value.image.c < config.crop_confidence_threshold: reasons.append("low_crop_confidence")
    if crop is None or not validate_crop(crop): reasons.append("invalid_crop")
    return reasons


def _reconcile_classification(value: CompactExtraction, context: dict) -> None:
    region_comments = [item for item in value.regions if item.t == "comment"]
    known = {item.v.strip().lower() for item in value.comments}
    for region in region_comments:
        normalized = region.v.strip().lower()
        if normalized not in known:
            value.comments.append(Comment(t="customer", n=None, v=region.v, c=region.c, b=region.b)); known.add(normalized)
    customer_texts = {item.v.strip().lower() for item in value.comments if item.v.strip()}
    owner_identity = " ".join(filter(None, (value.acct.v, value.handle.v))).lower()
    retained_replies = []
    for item in value.replies:
        normalized = item.v.strip().lower()
        duplicates_customer = any(normalized in text or text in normalized for text in customer_texts)
        wrong_author = bool(item.n and item.n.strip().lower() not in owner_identity)
        if item.t == "customer" or duplicates_customer or wrong_author:
            if not duplicates_customer:
                item.t = "customer"; value.comments.append(item); customer_texts.add(normalized)
        else: retained_replies.append(item)
    value.replies = retained_replies
    if value.caption.v:
        caption = value.caption.v
        trailing_comment = re.search(r"(?:^|\n|(?<=[.!?])\s+)([A-Za-z][A-Za-z0-9_. ]{0,30}):\s+([^#\n]+)$", caption)
        if trailing_comment:
            full_text = trailing_comment.group(0).strip()
            value.comments.append(Comment(t="customer", n=trailing_comment.group(1), v=trailing_comment.group(2).strip(), c=value.caption.c, b=None))
            caption = caption[:trailing_comment.start()].strip()
        for item in value.comments:
            if item.v and item.v.lower() in caption.lower():
                start = caption.lower().find(item.v.lower()); caption = (caption[:start] + caption[start + len(item.v):]).strip(" \n:-")
        value.caption.v = caption or None
    for field in (value.loc, value.address):
        if field.v and field.b is None:
            field.v = None; field.c = 0.0


def _remap_result_boxes(result: PickerExtractionResultV1, prepared: PreparedScreenshot) -> None:

async def extract_screenshot(*, image_bytes: bytes | None = None, context=None, request: PickerExtractionRequestV1 | None = None,
                             config: PickerConfig | None = None, dependencies: PickerDependencies | None = None,
                             prepared: PreparedScreenshot | None = None,
                             precomputed_hashes: tuple[str, str] | None = None) -> PickerExtractionResultV1:
    config = config or PickerConfig()
    dependencies = dependencies or PickerDependencies()
    if request is None:
        request = PickerExtractionRequestV1(image_bytes=image_bytes or b"", context=context or {})
    if not request.request_id:
        request.request_id = dependencies.id_generator()
    total_started = time.perf_counter()
    prepared = prepared or normalize_screenshot(request.image_bytes, config)
    hash_started = time.perf_counter()
    exact_hash, phash = precomputed_hashes or calculate_hashes(request.image_bytes, prepared.original)
    hashing_ms = 0 if precomputed_hashes else round((time.perf_counter() - hash_started) * 1000)
    candidate_started = time.perf_counter()
    provider_context = request.context.model_dump(mode="json")
    candidate_ms = round((time.perf_counter() - candidate_started) * 1000)
    provider = dependencies.vision_provider or OpenAIVisionProvider(config)
    reasons: list[str] = []
    first_response: ProviderResponse | None = None
    try:
        response = await provider.extract(image_bytes=prepared.normalized_bytes, context=provider_context, request_id=request.request_id,
            model=config.fast_vision_model, detail=config.image_detail, maximum_output_tokens=config.maximum_output_tokens)
        first_response = response
        reasons = _fallback_reasons(response.extraction, config)
    except PickerStructuredOutputError:
        reasons = ["structured_validation_failed"]
        response = None
    if reasons and config.fallback_enabled:
        fallback_bytes, fallback_size = encode_for_edge(prepared, config.fallback_maximum_edge, config.normalization_quality)
        response = await provider.extract(image_bytes=fallback_bytes, context=provider_context, request_id=request.request_id + "-fallback",
            model=config.fallback_vision_model, detail=config.fallback_image_detail, maximum_output_tokens=config.maximum_output_tokens)
        prepared.normalized_size = fallback_size
        if first_response:
            response.connection_ms += first_response.connection_ms; response.response_ms += first_response.response_ms
            response.parse_ms += first_response.parse_ms; response.input_tokens += first_response.input_tokens
            response.output_tokens += first_response.output_tokens; response.request_bytes += first_response.request_bytes
    if response is None:
        raise PickerStructuredOutputError("Picker extraction failed structured validation and fallback is disabled")
    _reconcile_classification(response.extraction, provider_context)
    metrics = PickerProcessingMetricsV1(
        normalization_ms=prepared.normalization_ms, hashing_ms=hashing_ms, candidate_preparation_ms=candidate_ms,
        openai_connection_ms=response.connection_ms, model_response_ms=response.response_ms, structured_parse_ms=response.parse_ms,
        total_ms=round((time.perf_counter() - total_started) * 1000), input_tokens=response.input_tokens,
        output_tokens=response.output_tokens, request_bytes=response.request_bytes, image_detail=response.image_detail,
        fallback_used=bool(reasons and config.fallback_enabled), fallback_reasons=reasons,
    )
    result = _to_result(response.extraction, request, prepared, response, exact_hash, phash, config, dependencies, metrics, reasons)
    if dependencies.metrics_recorder:
        dependencies.metrics_recorder("picker.extraction.total_ms", metrics.total_ms, {"model": response.model, "fallback": metrics.fallback_used})
    return result


async def extract_screenshots_bulk(request: PickerBulkExtractionRequestV1, *, config: PickerConfig | None = None,

/// Actor isolation keeps image decoding and Vision work off the main actor.
actor LocalOCRService {
    func recognize(_ data: Data) async throws -> [RecognizedTextItem] {
        try Task.checkCancellation()
        guard let source = CGImageSourceCreateWithData(data as CFData, nil),
              let image = CGImageSourceCreateImageAtIndex(source, 0, nil) else {
            throw LocalOCRError.unsupportedImage
        }
        let properties = CGImageSourceCopyPropertiesAtIndex(source, 0, nil) as? [CFString: Any]
        let rawOrientation = (properties?[kCGImagePropertyOrientation] as? NSNumber)?.uint32Value ?? 1
        let orientation = CGImagePropertyOrientation(rawValue: rawOrientation) ?? .up
        let request = VNRecognizeTextRequest()
        request.recognitionLevel = .accurate
        request.usesLanguageCorrection = true
        return try await withTaskCancellationHandler {
            try Task.checkCancellation()
            try VNImageRequestHandler(cgImage: image, orientation: orientation).perform([request])
            try Task.checkCancellation()
            return (request.results ?? []).compactMap { observation in
                guard let candidate = observation.topCandidates(1).first else { return nil }
                let box = observation.boundingBox
                return RecognizedTextItem(
                    text: candidate.string,
                    confidence: candidate.confidence,
                    boundingBox: TextBoundingBox(x: Double(box.minX), y: Double(box.minY),
                                                width: Double(box.width), height: Double(box.height))
                )
            }
        } onCancel: {
            request.cancel()
        }
    }
}

enum PickerSessionState: String {
    case idle, sourceSelected, processing, reviewing, completed, cancelled
}

@MainActor
final class PickerSession: ObservableObject {
    @Published private(set) var id = UUID()
    @Published private(set) var state: PickerSessionState = .idle
    @Published private(set) var sourceImageData: Data?
    @Published private(set) var observations: [RecognizedTextItem] = []
    @Published private(set) var cleanupFailed = false
    private let temporaryFiles: SessionTemporaryFiles

    init(temporaryFiles: SessionTemporaryFiles = SessionTemporaryFiles()) {
        self.temporaryFiles = temporaryFiles
        do { try temporaryFiles.removeAbandonedFiles() }
        catch { cleanupFailed = true }
    }

    /// Invalidates outstanding imports, including those not yet providing image data.
    @discardableResult
    func begin() -> UUID {
        purge()
        id = UUID()
        state = .idle
        return id
    }

    func selectSource(_ data: Data, for token: UUID) {
        guard token == id, state == .idle else { return }
        sourceImageData = data
        state = .sourceSelected
    }

    func startProcessing(for token: UUID) {
        guard token == id, state == .sourceSelected else { return }
        state = .processing
    }

    func review(_ items: [RecognizedTextItem], for token: UUID) {
        guard token == id, state == .processing else { return }
        sourceImageData = nil
        observations = items
        state = .reviewing
    }

    func complete() {
        guard state == .reviewing else { return }
        purge()
        state = .completed
    }

    func cancel() {
        purge()
        state = .cancelled
    }

    /// Releases our references; does not promise physical memory zeroization.
    func purge() {
        id = UUID() // Late asynchronous results can no longer populate this session.
        sourceImageData = nil
        observations.removeAll(keepingCapacity: false)
        state = .idle
        do {
            try temporaryFiles.removeAbandonedFiles()
            cleanupFailed = false
        } catch {
            cleanupFailed = true
        }
    }
}

Overview and problem

Picker converts screenshots and recordings into structured evidence for other applications. It gives host systems a reusable way to capture visible fields, content crops, and provenance before they make domain decisions.

Architecture and integrations

The Python package uses Pydantic contracts, Pillow image processing, HTTPX provider adapters, and local FFmpeg tools. Screenshot requests can use a configured OpenAI vision provider. Recording analysis samples frames and groups likely candidates. The separate SwiftUI and Vision iOS prototype explores local OCR.

Implemented capabilities and safeguards

Picker supports individual and bulk screenshot workflows, exact and perceptual hashes, configurable fallback, and bounded recording analysis. The consuming application owns access, persistence, and review. Provider-backed extraction sends image data to the configured provider. Recording candidates and recognized fields remain subject to review.

Engineering challenge

Screenshots and recordings can contain duplicate frames, transient overlays, and incomplete context. The extraction contract carries confidence and provenance so the host can review uncertain results.
